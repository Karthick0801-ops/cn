Q1. GRADE CONVERSION SERVICE – TCP
================================================
SHORT QUESTION:
Client sends a score (0–100). Server returns A/B/C/D/F. Invalid score gives
"Invalid score". Session continues until client sends "exit".

SERVER.C
--------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s, c;
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    char buf[100], result[100];

    s = socket(AF_INET, SOCK_STREAM, 0);
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(8080);
    bind(s, (struct sockaddr *)&server, sizeof(server));
    listen(s, 5);
    printf("Server waiting...\n");
    c = accept(s, (struct sockaddr *)&client, &len);

    while (1) {
        memset(buf, 0, sizeof(buf));
        recv(c, buf, sizeof(buf), 0);
        if (strcmp(buf, "exit") == 0) break;

        float score = atof(buf);
        if (score < 0 || score > 100) strcpy(result, "Invalid score");
        else if (score >= 90) strcpy(result, "Grade A");
        else if (score >= 80) strcpy(result, "Grade B");
        else if (score >= 70) strcpy(result, "Grade C");
        else if (score >= 60) strcpy(result, "Grade D");
        else strcpy(result, "Grade F");

        send(c, result, strlen(result) + 1, 0);
    }
    close(c); close(s);
    return 0;
}

--------

================================================
SHORT QUESTION:
Client sends distance and Day/Night. Server validates distance and flag,
then computes Day = 30 + 12*distance or Night = 30 + 18*distance.

#include <stdio.h>

Q9. STOP-AND-WAIT ARQ – HOME AUTOMATION
================================================
SHORT QUESTION:
N commands are sent one at a time. Given timeout command indices, retry each
affected command once; if retry fails, mark it failed. Count retransmissions
and failed commands.

COMPLETE PROGRAM
----------------
#include <stdio.h>

int main() {
    int N, K, timeout[100];
    int retransmissions = 0, failed = 0;

    printf("Enter number of commands: "); scanf("%d", &N);
    printf("Enter number of timeout commands: "); scanf("%d", &K);
    printf("Enter timeout command indices:\n");
    for (int i = 0; i < K; i++) scanf("%d", &timeout[i]);

    for (int cmd = 1; cmd <= N; cmd++) {
        int is_timeout = 0;
        for (int i = 0; i < K; i++)
            if (cmd == timeout[i]) { is_timeout = 1; break; }

        printf("\nSending Command %d\n", cmd);
        if (!is_timeout) {
            printf("ACK received for Command %d\n", cmd);
        } else {
            printf("Timeout for Command %d\n", cmd);
            printf("Retransmitting Command %d\n", cmd);
            retransmissions++;
            printf("Retry failed for Command %d\n", cmd);
            failed++;
        }
    }

    printf("\nTotal retransmissions = %d\n", retransmissions);
    printf("Total failed commands = %d\n", failed);
    return 0;
}

Q11. CURRENCY EXCHANGE – TCP
================================================
SHORT QUESTION:
Client sends amount, source currency and target currency. Supported rates:
USD=83.0, EUR=90.0, GBP=105.0, JPY=0.56, INR=1.0.
Validate codes, same currency and amount. Convert through INR. For amount >=
1000, commission is 2% of converted amount; otherwise subtract the target-
currency equivalent of 50 INR. Transaction ID = TXN + code lengths + floor(amount).

SERVER.C
--------
#include <stdio.h>
#include <string.h>
#include <math.h>
#include <unistd.h>
#include <arpa/inet.h>

struct Currency { char code[4]; float rate; };

int main() {
    int s, c;
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    char buf[200], result[500];
    struct Currency cur[5] = {
        {"USD",83.0},{"EUR",90.0},{"GBP",105.0},{"JPY",0.56},{"INR",1.0}
    };

    s = socket(AF_INET, SOCK_STREAM, 0);
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(8080);
    bind(s, (struct sockaddr *)&server, sizeof(server));
    listen(s, 5);
    printf("Server waiting...\n");
    c = accept(s, (struct sockaddr *)&client, &len);

    recv(c, buf, sizeof(buf), 0);
    float amount;
    char source[10], target[10];
    sscanf(buf, "%f %s %s", &amount, source, target);

    float sr=0, tr=0;
    int sf=0, tf=0;
    for (int i=0;i<5;i++) {
        if (strcmp(source,cur[i].code)==0) { sr=cur[i].rate; sf=1; }
        if (strcmp(target,cur[i].code)==0) { tr=cur[i].rate; tf=1; }
    }

    if (!sf || !tf) strcpy(result,"Invalid currency code");
    else if (strcmp(source,target)==0) strcpy(result,"Source and target currency cannot be same");
    else if (amount<=0) strcpy(result,"Invalid amount");
    else {
        float inr = amount * sr;
        float converted = inr / tr;
        float commission = (amount >= 1000) ? converted * 0.02 : 50.0 / tr;
        float final_amount = converted - commission;
        int code_length = strlen(source) + strlen(target);
        int whole = (int)floor(amount);
        sprintf(result,
                "Converted Amount: %.2f %s\nCommission: %.2f %s\nTransaction ID: TXN%d%d\nFinal Amount: %.2f %s",
                converted,target,commission,target,code_length,whole,final_amount,target);
    }
    send(c,result,strlen(result)+1,0);
    close(c); close(s);
    return 0;
}

Q12. PRIORITY PACKAGE DISPATCH – SLIDING WINDOW
================================================
SHORT QUESTION:
Given T package priorities E/S and W, keep original window membership,
but inside each window send E packages first, then S packages. Print ACKs,
running counts and final totals.

COMPLETE PROGRAM
----------------
#include <stdio.h>

int main() {
    int T,W;
    char p[100];
    int express=0, standard=0;

    printf("Enter total packages: "); scanf("%d",&T);
    printf("Enter window size: "); scanf("%d",&W);
    printf("Enter priorities E/S:\n");
    for(int i=0;i<T;i++) scanf(" %c",&p[i]);

    for(int start=0;start<T;start+=W) {
        int end=start+W;
        if(end>T) end=T;
        printf("\nWindow:\n");

        for(int i=start;i<end;i++) {
            if(p[i]=='E') {
                printf("Sent Package %d - E\n",i+1);
                printf("ACK Package %d\n",i+1);
                express++;
            }
        }
        for(int i=start;i<end;i++) {
            if(p[i]=='S') {
                printf("Sent Package %d - S\n",i+1);
                printf("ACK Package %d\n",i+1);
                standard++;
            }
        }
        printf("Running Express = %d\n",express);
        printf("Running Standard = %d\n",standard);
    }

    printf("\nTotal Express = %d\n",express);
    printf("Total Standard = %d\n",standard);
    if(express>standard) printf("Express had more packages.\n");
    else if(standard>express) printf("Standard had more packages.\n");
    else printf("Both priorities are equal.\n");
    return 0;
}

    sscanf(buf,"%s %s",mode,value);

    if(strcmp(mode,"byid")==0) {
        if(!valid_id(value)) strcpy(result,"Invalid ID format");
        else {
            int found=0;
            for(int i=0;i<10;i++) if(strcmp(value,e[i].id)==0) {
                sprintf(result,"Name: %s, Department: %s",e[i].name,e[i].dept);
                found=1; break;
            }
            if(!found) strcpy(result,"ID not found");
        }
    }
    else if(strcmp(mode,"bydept")==0) {
        int valid=0;
        for(int i=0;i<4;i++) if(strcasecmp(value,dept[i])==0) { valid=1; break; }
        if(!valid) strcpy(result,"Invalid department");
        else {
            int count=0;
            sprintf(result,"Employees in %s:\n",value);
            for(int i=0;i<10;i++) if(strcasecmp(value,e[i].dept)==0) {
                strcat(result,e[i].name);
                strcat(result," ");
                count++;
            }
            char temp[50];
            sprintf(temp,"\nCount = %d",count);
            strcat(result,temp);
        }
    }
    else strcpy(result,"Invalid mode");

    sendto(s,result,strlen(result)+1,0,(struct sockaddr *)&client,len);
    close(s);
    return 0;
}

Q14. SATELLITE COMMAND RELAY – STOP-AND-WAIT
================================================
SHORT QUESTION:
Status 0 = normal delivery and ACK.
Status 1 = command lost; no receiver line; timeout; retry once and retry succeeds.
Status 2 = command arrives and is processed; ACK lost; timeout; retransmit;
receiver detects duplicate, does not reprocess, and resends ACK.
Count total retransmissions and separately lost-command/lost-ACK retries.

COMPLETE PROGRAM
----------------
#include <stdio.h>

int main() {
    int N,status[100];
    int retransmissions=0,lost_commands=0,lost_acks=0;

    printf("Enter number of commands: "); scanf("%d",&N);
    printf("Enter status for each command:\n");
    printf("0 = normal, 1 = command lost, 2 = ACK lost\n");
    for(int i=0;i<N;i++) scanf("%d",&status[i]);

    for(int i=0;i<N;i++) {
        int cmd=i+1;
        printf("\nSending Command %d\n",cmd);

        if(status[i]==0) {
            printf("Receiver received Command %d\n",cmd);
            printf("Command %d processed\n",cmd);
            printf("ACK received for Command %d\n",cmd);
        }
        else if(status[i]==1) {
            printf("Command %d lost in transit\n",cmd);
            printf("Sender timeout\n");
            printf("Retransmitting Command %d\n",cmd);
            retransmissions++; lost_commands++;
            printf("Receiver received Command %d\n",cmd);
            printf("Command %d processed\n",cmd);
            printf("ACK received for Command %d\n",cmd);
        }
        else if(status[i]==2) {
            printf("Receiver received Command %d\n",cmd);
            printf("Command %d processed\n",cmd);
            printf("ACK lost\n");
            printf("Sender timeout\n");
            printf("Retransmitting Command %d\n",cmd);
            retransmissions++; lost_acks++;
            printf("Receiver detected duplicate Command %d\n",cmd);
            printf("Command %d NOT reprocessed\n",cmd);
            printf("ACK resent for Command %d\n",cmd);
            printf("ACK received by sender\n");
        }
        else printf("Invalid status for Command %d\n",cmd);
    }

    printf("\nTotal retransmissions = %d\n",retransmissions);
    printf("Due to lost commands = %d\n",lost_commands);
    printf("Due to lost ACKs = %d\n",lost_acks);
    return 0;
}
